# Data normalization and anti-normalization functions
def normalize_data(data):
    """Use the minimum-maximum normalization method to normalize the data. Formula (1)
    x* = (x - x_min)/(x_max - x_min)
    """
    min_vals = np.min(data, axis=0)
    max_vals = np.max(data, axis=0)
    # Avoid dividing by zero and use a very small value
    denominator = np.maximum(max_vals - min_vals, 1e-10)
    normalized_data = (data - min_vals) / denominator
    return normalized_data, min_vals, max_vals


def denormalize_data(normalized_data, min_vals, max_vals):
    """Anti-normalization processing
    x = x* * (x_max - x_min) + x_min
    """
    return normalized_data * (max_vals - min_vals) + min_vals


# Handle timestamp feature functions
def process_time_features(df):
    """Process the timestamp column and extract time features"""
    print("Processing timestamp features...")
    # Process timestamp column
    for col in df.columns:
        if pd.api.types.is_datetime64_any_dtype(df[col]):
            print(f"Processing timestamp column: {col}")
            # Use view() instead of astype() to avoid FutureWarning
            df[col] = pd.to_numeric(df[col].view('int64') // 10 ** 9)
            
            # If column name is time-related, extract time features
            if 'time' in col.lower():
                # Convert Unix timestamp back to datetime for feature extraction
                timestamps = pd.to_datetime(df[col], unit='s')
                df[f'{col}_hour'] = timestamps.dt.hour
                df[f'{col}_dayofweek'] = timestamps.dt.dayofweek
                df[f'{col}_month'] = timestamps.dt.month
                df[f'{col}_quarter'] = timestamps.dt.quarter
    
    return df


# Feature engineering function
def feature_engineering(df):
    """Perform feature engineering"""
    print("Performing feature engineering...")
    # Fill missing values
    df = df.fillna(method='ffill').fillna(method='bfill')
    
    # Extract feature columns
    features = [col for col in df.columns if col != 'PM2.5' and not col.startswith('time_')]
    
    try:
        # Add feature interactions
        if 'PM10' in df.columns and 'SO2' in df.columns:
            df['PM10_SO2'] = df['PM10'] * df['SO2']
        if 'NO2' in df.columns and 'O3' in df.columns:
            df['NO2_O3'] = df['NO2'] * df['O3']
        if 'highest temperature' in df.columns and 'lowest temperature' in df.columns:
            df['temperature difference'] = df['highest temperature'] - df['lowest temperature']
        
        # Fill NA values generated by feature engineering
        df = df.fillna(method='bfill').fillna(method='ffill')
        
    except Exception as e:
        print(f"Error during feature engineering: {str(e)}")
    
    return df


# Simplified data preprocessing method, using the same ratio as the second implementation
def simple_data_preprocessing(df, target_col='PM2.5', sequence_length=15):
    """Simplified data preprocessing, using 0.7225:0.1275:0.15 data split ratio, and normalize according to paper formula (1)"""
    print("Using simplified data preprocessing method...")
    
    # Separate features and target
    features = [col for col in df.columns if col != target_col]
    X = df[features].values
    y = df[target_col].values.reshape(-1, 1)
    
    # Use MinMaxScaler for normalization - according to paper formula (1)
    scaler_X = MinMaxScaler()
    scaler_y = MinMaxScaler()
    
    X_scaled = scaler_X.fit_transform(X)
    y_scaled = scaler_y.fit_transform(y)
    
    # Create sequence data
    if sequence_length > 1:
        X_scaled, y_scaled = create_sequences(X_scaled, y_scaled, sequence_length)
    
    # Split dataset - using 0.7225:0.1275:0.15 ratio
    n_samples = len(X_scaled)
    train_size = int(n_samples * 0.7225)
    val_size = int(n_samples * 0.1275)
    
    # First split training set
    X_train = X_scaled[:train_size]
    y_train = y_scaled[:train_size]
    
    # Then split validation set
    X_val = X_scaled[train_size:train_size+val_size]
    y_val = y_scaled[train_size:train_size+val_size]
    
    # Finally split test set
    X_test = X_scaled[train_size+val_size:]
    y_test = y_scaled[train_size+val_size:]
    
    print(f"Dataset split completed - Training set: {len(X_train)}, Validation set: {len(X_val)}, Test set: {len(X_test)}")
    
    return X_train, X_val, X_test, y_train, y_val, y_test, scaler_y

# Create time series data
def create_sequences(X, y, seq_length):
    """Create time series data"""
    Xs, ys = [], []
    for i in range(len(X) - seq_length):
        Xs.append(X[i:i + seq_length])
        ys.append(y[i + seq_length])
    return np.array(Xs), np.array(ys)


# Model evaluation function - simplified version
def evaluate_model(y_true, y_pred, scaler_y):
    """Evaluate model performance and return multiple evaluation metrics"""
    # Denormalize
    if scaler_y is not None:
        y_true = scaler_y.inverse_transform(y_true)
        y_pred = scaler_y.inverse_transform(y_pred)
    
    # Calculate evaluation metrics
    mse = mean_squared_error(y_true, y_pred)
    rmse = np.sqrt(mse)
    mae = mean_absolute_error(y_true, y_pred)
    r2 = r2_score(y_true, y_pred)
    
    # Calculate MAPE (Mean Absolute Percentage Error)
    epsilon = 1e-8  # Avoid division by zero
    mape = np.mean(np.abs((y_true - y_pred) / (np.abs(y_true) + epsilon))) * 100
    
    print(f'Evaluation metrics - MSE: {mse:.6f}, RMSE: {rmse:.6f}, MAE: {mae:.6f}, R²: {r2:.6f}, MAPE: {mape:.6f}%')
    
    return {
        'mse': mse,
        'rmse': rmse,
        'mae': mae,
        'r2': r2,
        'mape': mape
    }


# Correct BPNN model
class BPNN(nn.Module):
    """
    Backward propagation neural network - standard multilayer perceptron implementation
    
    This is a classic feedforward neural network, trained using backpropagation algorithm.
    It is suitable for various regression and classification tasks, but unlike RNN, it is not specifically designed for sequence data.
    """
    def __init__(self, input_size, hidden_size=128, output_size=1, dropout=0.2):
        """
        Initialize BPNN model
        
        Parameters:
            input_size: Dimension of input features
            hidden_size: Number of neurons in the hidden layer
            output_size: Output dimension
            dropout: Dropout rate, used to prevent overfitting
        """
        super(BPNN, self).__init__()
        
        # Model parameters
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.output_size = output_size
        self.dropout_rate = dropout
        
        # For sequence data, flatten
        self.flatten = nn.Flatten(start_dim=1)
        
        # Network layer definition
        # First layer: input layer to first hidden layer
        self.fc1 = nn.Linear(input_size, hidden_size)
        self.bn1 = nn.BatchNorm1d(hidden_size)
        self.dropout1 = nn.Dropout(dropout)
        
        # Second layer: first hidden layer to second hidden layer
        self.fc2 = nn.Linear(hidden_size, hidden_size // 2)
        self.bn2 = nn.BatchNorm1d(hidden_size // 2)
        self.dropout2 = nn.Dropout(dropout)
        
        # Output layer: second hidden layer to output layer
        self.fc3 = nn.Linear(hidden_size // 2, output_size)
        
        # Initialize weights
        self._init_weights()
        
    def _init_weights(self):
        """
        Use Xavier initialization for weights, which helps solve the problem of gradient vanishing/exploding
        """
        for m in self.modules():
            if isinstance(m, nn.Linear):
                nn.init.xavier_normal_(m.weight)
                if m.bias is not None:
                    nn.init.zeros_(m.bias)
    
    def forward(self, x, y_prev=None):
        """
        BPNN forward propagation
        
        Parameters:
            x: Input data, can be 2D [batch_size, features] or 3D [batch_size, seq_len, features]
            y_prev: Interface kept consistent with other models, but not used in BPNN
            
        Returns:
            output: Predicted result, shape [batch_size, output_size]
        """
        # Process input data - BPNN needs to flatten sequence data into feature vectors
        if len(x.shape) == 3:
            x = self.flatten(x)
        
        batch_size = x.size(0)
        
        # Forward propagation process
        # First layer
        x = self.fc1(x)
        if batch_size > 1:  # Batch normalization requires batch size > 1
            x = self.bn1(x)
        x = F.relu(x)
        x = self.dropout1(x)
        
        # Second layer
        x = self.fc2(x)
        if batch_size > 1:
            x = self.bn2(x)
        x = F.relu(x)
        x = self.dropout2(x)
        
        # Output layer
        x = self.fc3(x)
        
        return x


# Standard GRU model implementation
class GRU(nn.Module):
    def __init__(self, input_size, hidden_size=128, output_size=1, num_layers=2, dropout=0.2):
        """
        GRU network implementation - suitable for standard implementation of time series data
        
        Parameters:
            input_size: Dimension of input features
            hidden_size: Dimension of GRU hidden state
            output_size: Output dimension
            num_layers: Number of GRU layers
            dropout: Dropout rate
        """
        super(GRU, self).__init__()
        
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.num_layers = num_layers
        self.dropout_rate = dropout
        
        # Main GRU layer
        self.gru = nn.GRU(
            input_size=input_size,
            hidden_size=hidden_size,
            num_layers=num_layers,
            batch_first=True,
            dropout=0 if num_layers == 1 else dropout
        )
        
        # Batch normalization helps with training stability
        self.bn = nn.BatchNorm1d(hidden_size)
        
        # Output layer
        self.fc1 = nn.Linear(hidden_size, hidden_size//2)
        self.dropout = nn.Dropout(dropout)
        self.fc2 = nn.Linear(hidden_size//2, output_size)
        
        # Initialize weights
        self._init_weights()
    
    def _init_weights(self):
        """Initialize weights for GRU and linear layers"""
        for name, param in self.named_parameters():
            if 'weight_ih' in name:  # Weights from input to hidden layer
                nn.init.xavier_uniform_(param)
            elif 'weight_hh' in name:  # Weights from hidden layer to hidden layer
                nn.init.orthogonal_(param)  # Orthogonal initialization helps with gradient problem for long sequences
            elif 'bias' in name:  # Bias terms
                nn.init.zeros_(param)
            elif 'fc' in name and 'weight' in name:  # Weights for fully connected layers
                nn.init.xavier_normal_(param)
        
    def forward(self, x, y_prev=None):
        """
        GRU forward propagation - processing time series data
        
        Parameters:
            x: Input data, shape [batch_size, seq_length, input_size]
            y_prev: Parameter kept for API consistency, not used
            
        Returns:
            output: Predicted result, shape [batch_size, output_size]
        """
        # Ensure input is 3D [batch_size, seq_length, input_size]
        if len(x.shape) == 2:
            # If input is 2D, add time step dimension: [batch_size, features] -> [batch_size, 1, features]
            x = x.unsqueeze(1)
        
        # GRU processes sequence data - output: [batch_size, seq_len, hidden_size]
        # hidden: [num_layers, batch_size, hidden_size]
        output, hidden = self.gru(x)
        
        # Use output of the last time step
        last_output = output[:, -1, :]  # [batch_size, hidden_size]
        
        # Apply batch normalization - only when batch_size > 1
        if last_output.size(0) > 1:
            last_output = self.bn(last_output)
        
        # Output layer processing
        x = F.relu(self.fc1(last_output))
        x = self.dropout(x)
        x = self.fc2(x)
        
        return x


# Standard LSTM model implementation
class LSTM(nn.Module):
    def __init__(self, input_size, hidden_size=128, output_size=1, num_layers=2, dropout=0.2):
        """
        Long Short-Term Memory (LSTM) network implementation - suitable for standard implementation of long time series dependencies
        
        Parameters:
            input_size: Dimension of input features
            hidden_size: Dimension of LSTM hidden state
            output_size: Output dimension
            num_layers: Number of LSTM layers
            dropout: Dropout rate
        """
        super(LSTM, self).__init__()
        
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.num_layers = num_layers
        self.dropout_rate = dropout
        
        # LSTM layer - processing sequence data
        self.lstm = nn.LSTM(
            input_size=input_size,
            hidden_size=hidden_size,
            num_layers=num_layers,
            batch_first=True,
            dropout=0 if num_layers == 1 else dropout
        )
        
        # Batch normalization layer - helps with network training stability
        self.bn = nn.BatchNorm1d(hidden_size)
        
        # Output layer
        self.fc1 = nn.Linear(hidden_size, hidden_size//2)
        self.dropout = nn.Dropout(dropout)
        self.fc2 = nn.Linear(hidden_size//2, output_size)
        
        # Initialize weights
        self._init_weights()
    
    def _init_weights(self):
        """Initialize weights for LSTM and linear layers"""
        for name, param in self.named_parameters():
            if 'weight_ih' in name:  # Weights from input to hidden layer
                nn.init.xavier_uniform_(param)
            elif 'weight_hh' in name:  # Weights from hidden layer to hidden layer
                nn.init.orthogonal_(param)  # Orthogonal initialization helps with gradient problem for long sequences
            elif 'bias' in name:  # Bias terms
                # LSTM has four bias terms, initialize forget gate bias to 1, others to 0
                param.data.fill_(0)
                # f gate bias = 1
                if 'bias_ih' in name:
                    n = param.size(0)
                    param.data[n//4:n//2].fill_(1.0)
            elif 'fc' in name and 'weight' in name:  # Weights for fully connected layers
                nn.init.xavier_normal_(param)
        
    def forward(self, x, y_prev=None):
        """
        LSTM forward propagation - processing time series data
        
        Parameters:
            x: Input data, shape [batch_size, seq_length, input_size]
            y_prev: Parameter kept for API consistency, not used
            
        Returns:
            output: Predicted result, shape [batch_size, output_size]
        """
        # Ensure input is 3D [batch_size, seq_length, input_size]
        if len(x.shape) == 2:
            # If input is 2D, add time step dimension: [batch_size, features] -> [batch_size, 1, features]
            x = x.unsqueeze(1)
            
        # LSTM processes sequence data 
        # output: [batch_size, seq_len, hidden_size]
        # hidden: [num_layers, batch_size, hidden_size]
        # cell: [num_layers, batch_size, hidden_size]
        output, (hidden, cell) = self.lstm(x)
        
        # Use output of the last time step
        last_output = output[:, -1, :]  # [batch_size, hidden_size]
        
        # Apply batch normalization - only when batch_size > 1
        if last_output.size(0) > 1:
            last_output = self.bn(last_output)
        
        # Output layer processing
        x = F.relu(self.fc1(last_output))
        x = self.dropout(x)
        x = self.fc2(x)
        
        return x




# TDA-RNN model implementation
class TDA_RNN(nn.Module):
    def __init__(self, input_size, hidden_size=128, output_size=1, num_layers=2, dropout=0.2):
        super(TDA_RNN, self).__init__()
        
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.output_size = output_size
        self.num_layers = num_layers
        self.dropout = dropout
        
        # Set correct dropout value
        lstm_dropout = 0 if num_layers == 1 else dropout
        
        # Feature encoder LSTM - used for category attention
        self.feature_encoder_lstm = nn.LSTM(
            input_size=input_size,
            hidden_size=hidden_size,
            num_layers=num_layers,
            batch_first=True,
            dropout=lstm_dropout
        )
        
        # Temporal encoder LSTM - used for temporal attention
        self.temporal_encoder_lstm = nn.LSTM(
            input_size=input_size,
            hidden_size=hidden_size,
            num_layers=num_layers,
            batch_first=True,
            dropout=lstm_dropout
        )
        
        # Fusion encoder LSTM - used for processing features after attention weighting
        self.fusion_encoder_lstm = nn.LSTM(
            input_size=input_size,
            hidden_size=hidden_size,
            num_layers=num_layers,
            batch_first=True,
            dropout=lstm_dropout
        )
        
        # Decoder related layers - implemented strictly according to paper formula (11)(12)(13)(14)
        # W_T matrix in formula (11), used to fuse y_{t-1} and h_{t-1}
        self.W_T = nn.Linear(input_size + hidden_size, hidden_size)
        # R_T matrix in formula (11), used to process h_{t-1}
        self.R_T = nn.Linear(hidden_size, hidden_size)
        # Bias term b in formula (11)
        self.b = nn.Parameter(torch.zeros(hidden_size))
        
        # Decoder LSTM - implemented strictly according to paper formula (12)(13), number of layers same as encoder
        self.decoder_lstm = nn.LSTM(
            input_size=hidden_size,  # Input is h_t*
            hidden_size=hidden_size,
            num_layers=num_layers,  # Use same number of layers as encoder
            batch_first=True,
            dropout=lstm_dropout
        )
        
        # Output layer W_y and b_y in formula (14)
        self.W_y = nn.Linear(hidden_size, output_size)
        
        # Dropout layer
        self.dropout_layer = nn.Dropout(dropout)
        
        # Category attention parameters - implemented strictly according to paper formula (5)(6)
        self.W_1e = nn.Linear(2*hidden_size, hidden_size)  # W_{1,e}
        self.U_1e = nn.Linear(1, hidden_size)     # U_{1,e}
        self.v_1e = nn.Parameter(torch.FloatTensor(hidden_size).uniform_(-0.1, 0.1))  # v_{1,e}
        
        # Temporal attention parameters - implemented strictly according to paper formula (7)(8)
        self.W_2e = nn.Linear(2*hidden_size, hidden_size)  # W_{2,e}
        self.U_2e = nn.Linear(input_size, hidden_size)     # U_{2,e}
        self.v_2e = nn.Parameter(torch.FloatTensor(hidden_size).uniform_(-0.1, 0.1))  # v_{2,e}
        
        # Initialize weights
        self._init_weights()
    
    def _init_weights(self):
        """Initialize model weights"""
        for name, param in self.named_parameters():
            if 'weight' in name and 'lstm' not in name:
                nn.init.xavier_normal_(param)
    
    def compute_category_attention(self, x, h_states, s_states):
        """
        Calculate category attention - implemented strictly according to paper formula (5)(6)
        Args:
            x: Input sequence [batch_size, seq_length, input_size]
            h_states: Hidden states of all time steps [seq_length, batch_size, hidden_size]
            s_states: Cell states of all time steps [seq_length, batch_size, hidden_size]
        Returns:
            Category attention weights [batch_size, seq_length, input_size]
        """
        batch_size, seq_length, input_size = x.size()
        device = x.device
        
        # Initialize attention weights
        attention_weights = torch.zeros(batch_size, seq_length, input_size).to(device)
        
        # Calculate attention for each time step
        for t in range(seq_length):
            # Get hidden state and cell state of t-1 time step
            h_t_minus_1 = h_states[t-1] if t > 0 else h_states[0]  # [batch_size, hidden_size]
            s_t = s_states[t]  # [batch_size, hidden_size]
            
            # Concatenate states [h_{t-1}; s_t]
            h_s_concat = torch.cat([h_t_minus_1, s_t], dim=1)  # [batch_size, 2*hidden_size]
            
            # Calculate W_{1,e}[h_{t-1}; s_t]
            h_s_transform = self.W_1e(h_s_concat)  # [batch_size, hidden_size]
            
            # Initialize current time step's attention score
            attention_scores = torch.zeros(batch_size, input_size).to(device)
            
            # Calculate attention score for each feature k independently - strictly according to formula (5)
            for k in range(input_size):
                # Get value of feature k at time step t
                x_t_k = x[:, t, k]  # [batch_size]
                
                # Extend scalar feature to match hidden state dimension
                x_t_k = x_t_k.unsqueeze(1)  # [batch_size, 1]
                
                # Calculate U_{1,e}x_{t,k} - ensure output dimension matches hidden state
                x_k_transform = self.U_1e(x_t_k)  # [batch_size, hidden_size]
                
                # Calculate attention score e_{t,k} = v_{1,e}^T tanh(W_{1,e}[h_{t-1}; s_t] + U_{1,e}x_{t,k})
                attention = torch.tanh(h_s_transform + x_k_transform)  # [batch_size, hidden_size]
                e_t_k = torch.matmul(attention, self.v_1e)  # [batch_size]
                attention_scores[:, k] = e_t_k
            
            # Use Softmax to calculate attention weights α_{t,k} - strictly according to formula (6)
            # Normalize over feature dimension, ensuring that the sum of weights for all features k at each time step t is 1
            attention_weights[:, t, :] = F.softmax(attention_scores, dim=1)
        
        return attention_weights
    
    def compute_temporal_attention(self, x, h_states, s_states):
        """
        Calculate temporal attention - implemented strictly according to paper formula (7)(8)
        Args:
            x: Input sequence [batch_size, seq_length, input_size]
            h_states: Hidden states [num_layers, batch_size, hidden_size]
            s_states: Cell states [num_layers, batch_size, hidden_size]
        Returns:
            Temporal attention weights [batch_size, seq_length]
        """
        batch_size, seq_length, input_size = x.size()
        device = x.device
        
        # Initialize attention scores
        attention_scores = torch.zeros(batch_size, seq_length).to(device)
        
        # Get state of last layer
        h_last = h_states[-1]  # [batch_size, hidden_size]
        s_last = s_states[-1]  # [batch_size, hidden_size]
        
        # Concatenate hidden states and cell states [h_{k-1}; s_{k-1}]
        h_s_concat = torch.cat([h_last, s_last], dim=1)  # [batch_size, 2*hidden_size]
        h_s_transform = self.W_2e(h_s_concat)  # [batch_size, hidden_size]
        
        # Calculate attention score for each time step t - strictly according to formula (7)
        for t in range(seq_length):
            # Get input of time step t
            x_t = x[:, t, :]  # [batch_size, input_size]
            x_t_transform = self.U_2e(x_t)  # [batch_size, hidden_size]
            
            # Calculate attention score e_t = v_{2,e}^T tanh(W_{2,e}[h_{k-1}; s_{k-1}] + U_{2,e}x_t)
            attention = torch.tanh(h_s_transform + x_t_transform)  # [batch_size, hidden_size]
            e_t = torch.matmul(attention, self.v_2e)  # [batch_size]
            attention_scores[:, t] = e_t
        
        # Use Softmax to get attention weights β_t - strictly according to formula (8)
        attention_weights = F.softmax(attention_scores, dim=1)  # [batch_size, seq_length]
        
        return attention_weights
    
    def forward(self, x, y_prev=None, y_history=None):
        """
        TDA-RNN forward propagation - implemented strictly according to paper requirements
        Args:
            x: Input sequence [batch_size, seq_length, input_size]
            y_prev: Predicted value of previous time step [batch_size, output_size]
            y_history: Sequence of historical data [batch_size, seq_length, input_size]
        """
        if len(x.shape) == 2:
            x = x.unsqueeze(1)
        
        batch_size, seq_length, input_size = x.size()
        device = x.device
        
        # If historical data is not provided, use input sequence as historical data
        if y_history is None:
            y_history = x
        
        # 1. Feature encoding and category attention
        # Initialize LSTM state
        h_t = torch.zeros(self.num_layers, batch_size, self.hidden_size).to(device)
        s_t = torch.zeros(self.num_layers, batch_size, self.hidden_size).to(device)
        
        # Store state of all time steps
        h_states = []
        s_states = []
        
        # Run LSTM for each time step
        for t in range(seq_length):
            _, (h_t, s_t) = self.feature_encoder_lstm(
                x[:, t:t+1, :],
                (h_t, s_t)
            )
            h_states.append(h_t[-1])
            s_states.append(s_t[-1])
        
        # Convert state list to tensor
        h_states = torch.stack(h_states)  # [seq_length, batch_size, hidden_size]
        s_states = torch.stack(s_states)  # [seq_length, batch_size, hidden_size]
        
        # Calculate category attention weights
        category_attention = self.compute_category_attention(x, h_states, s_states)
        
        # 2. Temporal encoding and temporal attention
        temporal_output, (temporal_h, temporal_s) = self.temporal_encoder_lstm(x)
        temporal_attention = self.compute_temporal_attention(x, temporal_h, temporal_s)
        
        # 3. Attention fusion - implemented strictly according to paper formula (9)
        temporal_attention = temporal_attention.unsqueeze(-1).expand(-1, -1, input_size)
        combined_attention = category_attention + temporal_attention
        weighted_input = x * combined_attention
        
        # 4. Feature encoding - implemented strictly according to paper formula (10)
        fusion_output, (fusion_h, fusion_c) = self.fusion_encoder_lstm(weighted_input)
        
        # 5. Decoder processing - implemented strictly according to paper formula (11)(12)(13)(14)
        if y_prev is None:
            y_prev = torch.zeros(batch_size, self.output_size).to(device)
        
        # Get hidden state of last time step
        h_t_minus_1 = fusion_output[:, -1, :]  # [batch_size, hidden_size]
        
        # Use historical data of last time step
        y_hist_last = y_history[:, -1, :]  # [batch_size, input_size]
        
        # Formula (11): h_t* = W_T[y_{t-1}; h_{t-1}] + R_T h_{t-1} + b
        concat_input = torch.cat([y_hist_last, h_t_minus_1], dim=1)  # [batch_size, input_size + hidden_size]
        h_t_star = self.W_T(concat_input) + self.R_T(h_t_minus_1) + self.b
        h_t_star = torch.tanh(h_t_star)
        h_t_star = self.dropout_layer(h_t_star)
        
        # Formula (12)(13): Use LSTM to update hidden state
        # Extend h_t* to sequence form
        h_t_star = h_t_star.unsqueeze(1)  # [batch_size, 1, hidden_size]
        decoder_output, (decoder_h, decoder_c) = self.decoder_lstm(h_t_star)
        
        # Formula (14): Y = W_y h_t + b_y
        final_hidden = decoder_output[:, -1, :]  # [batch_size, hidden_size]
        final_hidden = self.dropout_layer(final_hidden)
        predictions = self.W_y(final_hidden)  # [batch_size, output_size]
        
        return predictions


# Position encoding class
class PositionalEncoding(nn.Module):
    def __init__(self, d_model, dropout=0.1, max_len=5000):
        super(PositionalEncoding, self).__init__()
        self.dropout = nn.Dropout(p=dropout)
        self.d_model = d_model  # Add d_model attribute

        pe = torch.zeros(max_len, d_model)
        position = torch.arange(0, max_len, dtype=torch.float).unsqueeze(1)
        div_term = torch.exp(
            torch.arange(0, d_model, 2).float() * (-math.log(10000.0) / d_model)
        )
        pe[:, 0::2] = torch.sin(position * div_term)
        pe[:, 1::2] = torch.cos(position * div_term)
        pe = pe.unsqueeze(0)
        self.register_buffer('pe', pe)
        
        # Add a linear layer for dimension transformation
        self.dim_mapper = None

    def forward(self, x):
        # Ensure input tensor's last dimension matches dimension of positional encoding
        if x.size(-1) != self.d_model:
            # If this is the first time encountering a dimension mismatch, create a linear transformation layer
            if self.dim_mapper is None or self.dim_mapper.in_features != x.size(-1):
                self.dim_mapper = nn.Linear(x.size(-1), self.d_model).to(x.device)
            # Use linear projection to adjust dimension
            x = self.dim_mapper(x)
        
        # Add positional encoding
        x = x + self.pe[:, :x.size(1)]
        return self.dropout(x)


# Modified simplified training function
def train_model_simple(model, X_train, y_train, X_val, y_val, X_test, y_test, 
                      scaler_y=None, epochs=200, lr=0.001, batch_size=32, weight_decay=1e-5):
    """Simplified version of model training function - increased weight decay and more training rounds"""
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    
    # Convert NumPy array to PyTorch tensor
    X_train_tensor = torch.FloatTensor(X_train).to(device)
    y_train_tensor = torch.FloatTensor(y_train).to(device)
    X_val_tensor = torch.FloatTensor(X_val).to(device)
    y_val_tensor = torch.FloatTensor(y_val).to(device)
    X_test_tensor = torch.FloatTensor(X_test).to(device)
    y_test_tensor = torch.FloatTensor(y_test).to(device)
    
    # For processing of neural network models
    model = model.to(device)
    
    # Define loss function
    criterion = nn.MSELoss()
    
    optimizer = torch.optim.Adam(model.parameters(), lr=lr, weight_decay=weight_decay)
    
    scheduler = torch.optim.lr_scheduler.ReduceLROnPlateau(optimizer, mode='min', factor=0.5, patience=10, min_lr=1e-6)
    
    # Training loop
    best_val_loss = float('inf')
    best_model = None
    patience = 20  # Increased patience value
    patience_counter = 0
    
    for epoch in range(epochs):
        model.train()
        
        # Create data loader
        train_dataset = torch.utils.data.TensorDataset(X_train_tensor, y_train_tensor)
        train_loader = torch.utils.data.DataLoader(train_dataset, batch_size=batch_size, shuffle=True)
        
        # Train one round
        train_loss = 0
        for batch_X, batch_y in train_loader:
            optimizer.zero_grad()
            
            # Forward propagation
            outputs = model(batch_X)
            loss = criterion(outputs, batch_y)
            
            # Backpropagation
            loss.backward()
            # Gradient clipping to prevent gradient explosion
            torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)
            optimizer.step()
            
            train_loss += loss.item() * batch_X.size(0)
        
        train_loss /= len(train_dataset)
        
        # Validation
        model.eval()
        with torch.no_grad():
            val_outputs = model(X_val_tensor)
            val_loss = criterion(val_outputs, y_val_tensor)
            
            # Print status
            print(f'Epoch {epoch+1}/{epochs}, training loss: {train_loss:.6f}, validation loss: {val_loss.item():.6f}')
            
            # Learning rate adjustment
            scheduler.step(val_loss)
            
            # Early stopping
            if val_loss < best_val_loss:
                best_val_loss = val_loss
                best_model = model.state_dict().copy()
                patience_counter = 0
            else:
                patience_counter += 1
                if patience_counter >= patience:
                    print(f'Early stopping: validation loss did not improve for {patience} rounds')
                    break
    
    # Use best model for testing
    if best_model is not None:
        model.load_state_dict(best_model)
    
    # Test evaluation
    model.eval()
    with torch.no_grad():
        test_outputs = model(X_test_tensor)
        
        # Convert to NumPy array for evaluation
        test_pred = test_outputs.cpu().numpy()
        test_true = y_test_tensor.cpu().numpy()
        
        # Calculate evaluation metrics
        results = evaluate_model(test_true, test_pred, scaler_y)
        
        return model, results

    # Data preprocessing
    df = process_time_features(df)
    df = feature_engineering(df)
    
    # Extract features and target
    features = [col for col in df.columns if col != target_col]
    X = df[features].values
    y = df[target_col].values.reshape(-1, 1)
    
    # Initialize K-fold cross-validation
    kf = KFold(n_splits=n_splits, shuffle=True, random_state=42)

    for fold, (train_idx, test_idx) in enumerate(kf.split(X)):

# MA model implementation
class MAModel(nn.Module):
    def __init__(self, window_size=15, output_size=1, noise_std=0.1, noise_mean=0.0, use_noise=False):
        super(MAModel, self).__init__()
        self.window_size = window_size
        self.output_size = output_size
        self.noise_std = noise_std
        self.noise_mean = noise_mean
        self.use_noise = use_noise
        self.device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
        self.data_stats = {'mean': None, 'std': None}
    
    def update_data_stats(self, x):
        """Update data statistics"""
        if self.data_stats['mean'] is None:
            self.data_stats['mean'] = torch.mean(x, dim=0, keepdim=True)
            self.data_stats['std'] = torch.std(x, dim=0, keepdim=True)
    
    def add_noise(self, x):
        """Add scaled Gaussian noise to input data"""
        if not self.use_noise:
            return x
            
        # Update data statistics
        self.update_data_stats(x)
        
        # Generate noise with the same shape as input
        noise = torch.randn_like(x)
        
        # Scale noise based on data standard deviation
        scaled_noise = noise * self.noise_std * self.data_stats['std']
        
        # Add noise
        return x + scaled_noise
    
    def calculate_moving_average(self, x):
        device = x.device
        
        if len(x.shape) == 2:
            x = x.unsqueeze(1)
            
        batch_size, seq_len, features = x.shape
        effective_window = min(self.window_size, seq_len)
        
        # Take the first feature of the last effective_window time steps
        recent_values = x[:, -effective_window:, 0]
        
        # Add noise
        recent_values = self.add_noise(recent_values)
        
        # Calculate simple moving average
        moving_avg = torch.mean(recent_values, dim=1, keepdim=True)
        
        return moving_avg
        
    def weighted_moving_average(self, x, weights=None):
        if len(x.shape) == 2:
            x = x.unsqueeze(1)
            
        batch_size, seq_len, features = x.shape
        device = x.device
        effective_window = min(self.window_size, seq_len)
        
        recent_values = x[:, -effective_window:, 0]
        recent_values = self.add_noise(recent_values)
        
        if weights is None:
            weights = torch.linspace(0.5, 1.0, effective_window).to(device)
            weights = weights / weights.sum()
        
        weighted_avg = torch.sum(recent_values * weights.unsqueeze(0), dim=1, keepdim=True)
        
        return weighted_avg
    
    def exponential_moving_average(self, x, alpha=0.3):
        if len(x.shape) == 2:
            x = x.unsqueeze(1)
            
        batch_size, seq_len, features = x.shape
        effective_window = min(self.window_size, seq_len)
        
        recent_values = x[:, -effective_window:, 0]
        recent_values = self.add_noise(recent_values)
        
        ema = torch.zeros(batch_size, 1).to(x.device)
        
        if effective_window > 0:
            ema = recent_values[:, 0].unsqueeze(1)
            
            for t in range(1, effective_window):
                ema = alpha * recent_values[:, t].unsqueeze(1) + (1 - alpha) * ema
        
        return ema
    
    def forward(self, x, y_prev=None, method='simple'):
        self.device = x.device

        if method == 'weighted':
            return self.weighted_moving_average(x)
        elif method == 'exponential':
            return self.exponential_moving_average(x)
        else:
            return self.calculate_moving_average(x)
